# Riskline Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Riskline, a static page and a CLI that turn one unified diff into a card of changes that can break a running app, plus a pull request description copied from those findings.

**Architecture:** `packages/rules` parses a unified diff and runs deterministic rules in the browser and in Node. `apps/web` is a Next.js page that calls those functions locally and never uploads the diff. `apps/cli` reads a file, stdin, or `git diff` and prints the same Markdown.

**Tech Stack:** pnpm 10.28.1 workspaces, TypeScript, Vitest 5, Next.js 16.4.2, React 19.3.0, tsx for the CLI.

## Global Constraints

- Diff size limit is `2 * 1024 * 1024` bytes. Over that, do not parse.
- Website empty message: `Paste a unified diff.`
- Website and CLI not-a-diff message: `This is not a unified diff. Use git diff or git diff <base>...HEAD.`
- Website oversize message: `This diff is over 2 MB. Run the CLI on the branch.`
- CLI no-diff message: `No diff to read. Pipe a diff, pass --file, or run inside a git repository.`
- CLI oversize message: `This diff is over 2 MB.`
- CLI missing base message: `No base ref. Pass --base, or create main.`
- CLI unreadable file message: `Cannot read <path>.`
- A possible secret is summarized as `Possible secret added.` The matched text is never copied into a summary or the Markdown.
- Findings sort by file path, then line, then rule id.
- The diff is not uploaded. v1 has no accounts, no history, no GitHub Action, and no model-written summary.
- The CLI command name is `riskline`. Stdout is the Markdown card only. Exit `0` when a card is printed, `1` for empty, invalid, git, or missing-base failures, `2` when the diff is over 2 MB.
- The copy button copies from `## What can break` through the end. It does not copy `# Riskline` or `## Pull request`.

```typescript
import { isUnifiedDiff } from "./parse.js";

export const MAX_DIFF_BYTES = 2 * 1024 * 1024;

export type ScreenFailure = "empty" | "not-diff" | "too-large";

export type ScreenResult =
  | { ok: true; diff: string }
  | { ok: false; error: ScreenFailure; message: string };

const messages: Record<ScreenFailure, string> = {
  empty: "Paste a unified diff.",
  "not-diff": "This is not a unified diff. Use git diff or git diff <base>...HEAD.",
  "too-large": "This diff is over 2 MB. Run the CLI on the branch.",
};

function byteLength(text: string): number {
  return new TextEncoder().encode(text).length;
}

export function screenDiff(input: string): ScreenResult {
  if (byteLength(input) > MAX_DIFF_BYTES) {
    return { ok: false, error: "too-large", message: messages["too-large"] };
  }
  if (input.trim() === "") {
    return { ok: false, error: "empty", message: messages.empty };
  }
  if (!isUnifiedDiff(input)) {
    return { ok: false, error: "not-diff", message: messages["not-diff"] };
  }
  return { ok: true, diff: input };
}
```

```typescript
type Mark = " " | "+" | "-";

export function diffFixture(file: {
  path: string;
  lines: Array<{ kind: Mark; text: string }>;
  oldStart?: number;
  newStart?: number;
  deleted?: boolean;
  created?: boolean;
}): string {
  const oldCount = file.lines.filter((line) => line.kind !== "+").length;
  const newCount = file.lines.filter((line) => line.kind !== "-").length;
  const oldStart = file.oldStart ?? 1;
  const newStart = file.newStart ?? 1;
  const oldHeader = file.created ? "/dev/null" : `a/${file.path}`;
  const newHeader = file.deleted ? "/dev/null" : `b/${file.path}`;
  const body = file.lines.map((line) => `${line.kind}${line.text}`).join("\n");
  return [
    `diff --git a/${file.path} b/${file.path}`,
    `--- ${oldHeader}`,
    `+++ ${newHeader}`,
    `@@ -${oldStart},${oldCount} +${newStart},${newCount} @@`,
    body,
    "",
  ].join("\n");
}

export function diffs(...parts: string[]): string {
  return parts.join("");
}
```

Replace `packages/rules/src/parse.ts`:

```typescript
export type DiffRow = {
  kind: "context" | "add" | "remove";
  text: string;
  oldLine: number | null;
  newLine: number | null;
};

export type DiffFile = {
  oldPath: string | null;
  newPath: string | null;
  binary: boolean;
  rows: DiffRow[];
};

export function pathOf(file: DiffFile): string {
  return file.newPath ?? file.oldPath ?? "(unknown)";
}

export function isUnifiedDiff(input: string): boolean {
  if (input.includes("diff --git ")) return true;
  const lines = input.split(/\r?\n/);
  let sawMinus = false;
  let sawPlus = false;
  let sawHunk = false;
  for (const line of lines) {
    if (line.startsWith("--- ")) sawMinus = true;
    else if (line.startsWith("+++ ")) sawPlus = true;
    else if (line.startsWith("@@")) sawHunk = true;
  }
  return sawMinus && sawPlus && sawHunk;
}

function stripGitPrefix(path: string): string {
  if (path.startsWith("a/") || path.startsWith("b/")) return path.slice(2);
  return path;
}

function nullIfDevNull(path: string): string | null {
  return path === "/dev/null" ? null : path;
}

function parseGitHeader(line: string): { oldPath: string | null; newPath: string | null } {
  const rest = line.slice("diff --git ".length);
  const quoted = /^"((?:\\.|[^"])*)"\s+"((?:\\.|[^"])*)"$/.exec(rest);
  if (quoted) {
    return {
      oldPath: nullIfDevNull(stripGitPrefix(quoted[1].replace(/\\"/g, '"'))),
      newPath: nullIfDevNull(stripGitPrefix(quoted[2].replace(/\\"/g, '"'))),
    };
  }
  if (rest.startsWith("a/")) {
    const splitAt = rest.indexOf(" b/");
    if (splitAt !== -1) {
      return {
        oldPath: nullIfDevNull(stripGitPrefix(rest.slice(0, splitAt))),
        newPath: nullIfDevNull(stripGitPrefix(rest.slice(splitAt + 1))),
      };
    }
  }
  const [left, right] = rest.split(" ");
  return {
    oldPath: left ? nullIfDevNull(stripGitPrefix(left)) : null,
    newPath: right ? nullIfDevNull(stripGitPrefix(right)) : null,
  };
}

export function parseDiff(input: string): DiffFile[] {
  const lines = input.split(/\r?\n/);
  const files: DiffFile[] = [];
  let current: DiffFile | null = null;
  let oldLine = 0;
  let newLine = 0;
  let inHunk = false;
  let binaryPatch = false;

  for (const line of lines) {
    if (line.startsWith("diff --git ")) {
      binaryPatch = false;
      inHunk = false;
      const paths = parseGitHeader(line);
      current = { oldPath: paths.oldPath, newPath: paths.newPath, binary: false, rows: [] };
      files.push(current);
      continue;
    }

    if (!current && (line.startsWith("--- ") || line.startsWith("Binary files ") || line.startsWith("GIT binary patch"))) {
      current = { oldPath: null, newPath: null, binary: false, rows: [] };
      files.push(current);
    }

    if (!current) continue;

    if (line.startsWith("rename from ")) {
      current.oldPath = line.slice("rename from ".length).trim();
      continue;
    }
    if (line.startsWith("rename to ")) {
      current.newPath = line.slice("rename to ".length).trim();
      continue;
    }
    if (line.startsWith("Binary files ") || line.startsWith("GIT binary patch")) {
      current.binary = true;
      current.rows = [];
      binaryPatch = line.startsWith("GIT binary patch");
      inHunk = false;
      continue;
    }
    if (binaryPatch) continue;

    if (line.startsWith("--- ")) {
      const path = line.slice(4).split("\t")[0];
      current.oldPath = nullIfDevNull(path === "/dev/null" ? path : stripGitPrefix(path));
      inHunk = false;
      continue;
    }
    if (line.startsWith("+++ ")) {
      const path = line.slice(4).split("\t")[0];
      current.newPath = nullIfDevNull(path === "/dev/null" ? path : stripGitPrefix(path));
      inHunk = false;
      continue;
    }

    const hunk = /^@@ -(\d+)(?:,\d+)? \+(\d+)(?:,\d+)? @@/.exec(line);
    if (hunk) {
      oldLine = Number(hunk[1]);
      newLine = Number(hunk[2]);
      inHunk = true;
      continue;
    }
    if (!inHunk || line.startsWith("\\")) continue;

    if (line.startsWith("+")) {
      current.rows.push({ kind: "add", text: line.slice(1), oldLine: null, newLine });
      newLine += 1;
      continue;
    }
    if (line.startsWith("-")) {
      current.rows.push({ kind: "remove", text: line.slice(1), oldLine, newLine: null });
      oldLine += 1;
      continue;
    }
    if (line.startsWith(" ")) {
      current.rows.push({ kind: "context", text: line.slice(1), oldLine, newLine });
      oldLine += 1;
      newLine += 1;
    }
  }

  return files;
}
```

```typescript
import type { DiffFile } from "./parse.js";
import type { Card, Finding } from "./types.js";

function findingLine(finding: Finding): string {
  return `- \`${finding.file}\` line ${finding.line} — ${finding.summary}`;
}

export function pullRequestBody(findings: Finding[]): string {
  const breaks = findings.filter((finding) => finding.severity === "break");
  const checks = findings.filter((finding) => finding.severity === "check");
  const breakBlock = breaks.length
    ? breaks.map((finding) => `- ${finding.summary} (\`${finding.file}:${finding.line}\`)`).join("\n")
    : "None.";
  const checkBlock = checks.length
    ? checks.map((finding) => `- [ ] ${finding.summary} (\`${finding.file}:${finding.line}\`)`).join("\n")
    : "None.";
  return `## What can break\n${breakBlock}\n\n## Confirm before merge\n${checkBlock}\n`;
}

export function renderCard(files: DiffFile[], findings: Finding[]): Card {
  const breaks = findings.filter((finding) => finding.severity === "break");
  const checks = findings.filter((finding) => finding.severity === "check");
  const pullRequest = pullRequestBody(findings);
  const lines = [
    "# Riskline",
    "",
    `Files: ${files.length} · Break: ${breaks.length} · Check: ${checks.length}`,
    "",
  ];
  if (breaks.length === 0 && checks.length === 0) {
    lines.push("No break or check findings in this diff.", "");
  } else {
    if (breaks.length > 0) lines.push("## Break", ...breaks.map(findingLine), "");
    if (checks.length > 0) lines.push("## Check", ...checks.map(findingLine), "");
  }
  lines.push("## Pull request", "", pullRequest.trimEnd(), "");
  return {
    files: files.length,
    breakCount: breaks.length,
    checkCount: checks.length,
    findings,
    markdown: lines.join("\n"),
    pullRequest,
  };
}
```

```typescript
import type { Finding, Rule } from "../types.js";
import { filePath, rowLine } from "./lines.js";

const readers = [
  /process\.env\.([A-Z][A-Z0-9_]*)/g,
  /process\.env\[['"]([A-Z][A-Z0-9_]*)['"]\]/g,
  /os\.environ\[['"]([A-Z][A-Z0-9_]*)['"]\]/g,
  /os\.environ\.get\(['"]([A-Z][A-Z0-9_]*)['"]\)/g,
  /os\.Getenv\(['"]([A-Z][A-Z0-9_]*)['"]\)/g,
  /ENV\.fetch\(['"]([A-Z][A-Z0-9_]*)['"]\)/g,
  /ENV\[['"]([A-Z][A-Z0-9_]*)['"]\]/g,
];

function exampleHas(files: DiffFile[], name: string): boolean {
  const pattern = new RegExp(`(^|[^A-Z0-9_])${name}([^A-Z0-9_]|$)`);
  return files.some((file) => {
    const base = (file.newPath ?? file.oldPath ?? "").split("/").pop();
    if (base !== ".env.example" && base !== ".env.sample") return false;
    return file.rows.some((row) => row.kind === "add" && pattern.test(row.text));
  });
}

export const envWithoutExample: Rule = {
  id: "env-without-example",
  run(files: DiffFile[]): Finding[] {
    const seen = new Set<string>();
    const findings: Finding[] = [];
    for (const file of files) {
      for (const row of file.rows) {
        if (row.kind !== "add") continue;
        for (const reader of readers) {
          reader.lastIndex = 0;
          let match = reader.exec(row.text);
          while (match) {
            const name = match[1];
            if (!seen.has(name) && !exampleHas(files, name)) {
              seen.add(name);
              findings.push({
                severity: "check",
                rule: "env-without-example",
                file: filePath(file),
                line: rowLine(row),
                summary: `Env var ${name} is read and is not added to an example env file.`,
              });
            }
            match = reader.exec(row.text);
          }
        }
      }
    }
    return findings;
  },
```

```typescript
import type { DiffFile, DiffRow } from "../parse.js";
import { pathOf } from "../parse.js";
import type { Finding, Rule } from "../types.js";
import { rowLine } from "./lines.js";

const depSections = new Set(["dependencies", "devDependencies", "peerDependencies", "optionalDependencies"]);
const cargoTables = new Set(["dependencies", "dev-dependencies", "build-dependencies"]);

type Touch = { name: string; text: string; kind: "add" | "remove"; line: number };

function baseName(file: DiffFile): string {
  const path = pathOf(file);
  return path.split("/").pop() ?? path;
}

function toFindings(file: DiffFile, touches: Touch[]): Finding[] {
  const grouped = new Map<string, Touch[]>();
  for (const touch of touches) {
    const list = grouped.get(touch.name) ?? [];
    list.push(touch);
    grouped.set(touch.name, list);
  }
  const findings: Finding[] = [];
  for (const [name, list] of grouped) {
    const added = list.filter((item) => item.kind === "add");
    const removed = list.filter((item) => item.kind === "remove");
    if (added.length === 0) continue;
    const addedText = added.map((item) => item.text).sort().join("\n");
    const removedText = removed.map((item) => item.text).sort().join("\n");
    const summary = removed.length === 0 ? `Dependency ${name} was added.` : `Dependency ${name} changed version.`;
    if (removed.length !== 0 && addedText === removedText) continue;
    findings.push({
      severity: "check",
      rule: "dependency",
      file: pathOf(file),
      line: added[0].line,
      summary,
    });
  }
  return findings;
}

function changed(row: DiffRow): row is DiffRow & { kind: "add" | "remove" } {
  return row.kind === "add" || row.kind === "remove";
}

function packageJsonTouches(file: DiffFile): Touch[] {
  const touches: Touch[] = [];
  let section: string | null = null;
  for (const row of file.rows) {
    const header = /^\s*"([^"]+)"\s*:\s*[{[]/.exec(row.text);
    if (header) section = header[1];
    if (!changed(row) || !section || !depSections.has(section)) continue;
    const pkg = /^\s*"([^"]+)"\s*:\s*"([^"]*)"\s*,?\s*$/.exec(row.text);
    if (!pkg) continue;
    touches.push({ name: pkg[1], text: row.text.trim(), kind: row.kind, line: rowLine(row) });
  }
  return touches;
}

function cargoTouches(file: DiffFile): Touch[] {
  const touches: Touch[] = [];
  let section: string | null = null;
  for (const row of file.rows) {
    const table = /^\s*\[([^\]]+)\]\s*$/.exec(row.text);
    if (table) section = table[1];
    if (!changed(row) || !section || !cargoTables.has(section)) continue;
    const pkg = /^\s*([A-Za-z0-9_-]+)\s*=\s*(.*)$/.exec(row.text);
    if (!pkg) continue;
    touches.push({ name: pkg[1], text: row.text.trim(), kind: row.kind, line: rowLine(row) });
  }
  return touches;
}

function goTouches(file: DiffFile): Touch[] {
  const touches: Touch[] = [];
  for (const row of file.rows) {
    if (!changed(row)) continue;
    const match = /^\s*([^\s]+)\s+(v\d\S*)/.exec(row.text);
    if (!match) continue;
    touches.push({ name: match[1], text: row.text.trim(), kind: row.kind, line: rowLine(row) });
  }
  return touches;
}

function requirementName(text: string): string | null {
  const trimmed = text.trim();
  if (!trimmed || trimmed.startsWith("#") || trimmed.startsWith("-")) return null;
  const name = trimmed.split(/\[| |==|>=|<=|~=|!=/)[0];
  return name || null;
}

function requirementTouches(file: DiffFile): Touch[] {
  const touches: Touch[] = [];
  for (const row of file.rows) {
    if (!changed(row)) continue;
    const name = requirementName(row.text);
    if (!name) continue;
    touches.push({ name, text: row.text.trim(), kind: row.kind, line: rowLine(row) });
  }
  return touches;
}

function pepName(value: string): string {
  return value.split(/ |==|>=|<=|~=|!=/)[0];
}

function pyprojectTouches(file: DiffFile): Touch[] {
  const touches: Touch[] = [];
  let header: string | null = null;
  let inProjectDeps = false;
  for (const row of file.rows) {
    const table = /^\s*\[([^\]]+)\]\s*$/.exec(row.text);
    if (table) {
      header = table[1];
      inProjectDeps = false;
      continue;
    }
    const poetry = header === "tool.poetry.dependencies" || /^tool\.poetry\.group\.[^.]+\.dependencies$/.test(header ?? "");
    if (poetry && changed(row)) {
      const pkg = /^\s*([A-Za-z0-9_-]+)\s*=\s*(.*)$/.exec(row.text);
      if (pkg) touches.push({ name: pkg[1], text: row.text.trim(), kind: row.kind, line: rowLine(row) });
      continue;
    }
    if (header !== "project") continue;
    if (/^\s*dependencies\s*=\s*\[/.test(row.text)) inProjectDeps = true;
    if (!inProjectDeps) continue;
    if (changed(row)) {
      const quoted = /"([^"]+)"/g;
      let match = quoted.exec(row.text);
      while (match) {
        const name = pepName(match[1]);
        if (name) touches.push({ name, text: match[1], kind: row.kind, line: rowLine(row) });
        match = quoted.exec(row.text);
      }
    }
    if (row.text.includes("]")) inProjectDeps = false;
  }
  return touches;
}

export const dependency: Rule = {
  id: "dependency",
  run(files: DiffFile[]): Finding[] {
    const findings: Finding[] = [];
    for (const file of files) {
      const name = baseName(file);
      if (name === "package.json") findings.push(...toFindings(file, packageJsonTouches(file)));
      else if (name === "Cargo.toml") findings.push(...toFindings(file, cargoTouches(file)));
      else if (name === "go.mod") findings.push(...toFindings(file, goTouches(file)));
      else if (name === "requirements.txt") findings.push(...toFindings(file, requirementTouches(file)));
      else if (name === "pyproject.toml") findings.push(...toFindings(file, pyprojectTouches(file)));
    }
    return findings;
  },
};
```

```typescript
import type { DiffFile } from "../parse.js";
import type { Finding, Rule } from "../types.js";

export function runRules(files: DiffFile[], rules: Rule[]): Finding[] {
  const findings: Finding[] = [];
  for (const rule of rules) {
    try {
      findings.push(...rule.run(files));
    } catch {
      findings.push({
        severity: "check",
        rule: "rule-failed",
        file: "(rules)",
        line: 1,
        summary: `Rule ${rule.id} failed and was skipped.`,
      });
    }
  }
  findings.sort((a, b) => {
    const byFile = a.file.localeCompare(b.file);
    if (byFile !== 0) return byFile;
    if (a.line !== b.line) return a.line - b.line;
    return a.rule.localeCompare(b.rule);
  });
  return findings;
}
```

```typescript
import { spawnSync } from "node:child_process";
import { analyze, screenDiff } from "@riskline/rules";

export type Io = {
  cwd: string;
  isTTY: boolean;
  readStdin: () => string;
  readFile: (path: string) => string;
};

export type CliResult = { code: number; stdout: string; stderr: string };

const cliMessages = {
  empty: "No diff to read. Pipe a diff, pass --file, or run inside a git repository.\n",
  "not-diff": "This is not a unified diff. Use git diff or git diff <base>...HEAD.\n",
  "too-large": "This diff is over 2 MB.\n",
  "no-base": "No base ref. Pass --base, or create main.\n",
};

function fail(stderr: string, code = 1): CliResult {
  return { code, stdout: "", stderr };
}

function git(cwd: string, args: string[]) {
  return spawnSync("git", args, { cwd, encoding: "utf8", maxBuffer: 20 * 1024 * 1024 });
}

function readGitDiff(cwd: string, base: string | null): CliResult | { text: string } {
  let resolved = base;
  if (!resolved) {
    if (git(cwd, ["rev-parse", "--verify", "--quiet", "main"]).status === 0) resolved = "main";
    else if (git(cwd, ["rev-parse", "--verify", "--quiet", "master"]).status === 0) resolved = "master";
    else return fail(cliMessages["no-base"]);
  }
  const diff = git(cwd, ["diff", `${resolved}...HEAD`]);
  if (diff.status !== 0) return fail(diff.stderr || "git diff failed\n");
  return { text: diff.stdout };
}

export async function run(argv: string[], io: Io): Promise<CliResult> {
  let file: string | null = null;
  let base: string | null = null;
  for (let index = 0; index < argv.length; index += 1) {
    const arg = argv[index];
    if (arg === "--file") {
      const value = argv[index + 1];
      if (!value) return fail("Missing value for --file.\n");
      file = value;
      index += 1;
      continue;
    }
    if (arg === "--base") {
      const value = argv[index + 1];
      if (!value) return fail("Missing value for --base.\n");
      base = value;
      index += 1;
      continue;
    }
    return fail(`Unknown argument ${arg}.\n`);
  }

  let text = "";
  if (file) {
    try {
      text = io.readFile(file);
    } catch {
      return fail(`Cannot read ${file}.\n`);
    }
  } else if (!io.isTTY) {
    text = io.readStdin();
  } else {
    const inside = git(io.cwd, ["rev-parse", "--is-inside-work-tree"]);
    if (inside.status !== 0 || inside.stdout.trim() !== "true") return fail(cliMessages.empty);
    const diff = readGitDiff(io.cwd, base);
    if ("code" in diff) return diff;
    text = diff.text;
  }

  const screened = screenDiff(text);
  if (!screened.ok) {
    const code = screened.error === "too-large" ? 2 : 1;
    const stderr = screened.error === "empty" ? cliMessages.empty : screened.error === "too-large" ? cliMessages["too-large"] : cliMessages["not-diff"];
    return fail(stderr, code);
  }
  const card = analyze(screened.diff);
  return { code: 0, stdout: card.markdown.endsWith("\n") ? card.markdown : `${card.markdown}\n`, stderr: "" };
}
```

Create `apps/web/lib/check-diff.ts`:

```typescript
import { analyze, screenDiff, type Card } from "@riskline/rules";

export type CheckResult = { kind: "message"; message: string } | { kind: "card"; card: Card };

export function checkDiff(input: string): CheckResult {
  const screened = screenDiff(input);
  if (!screened.ok) return { kind: "message", message: screened.message };
  return { kind: "card", card: analyze(screened.diff) };
}
```

```tsx
"use client";

import { useState } from "react";
import { checkDiff, type CheckResult } from "../lib/check-diff";

export default function Page() {
  const [diff, setDiff] = useState("");
  const [result, setResult] = useState<CheckResult | null>(null);
  const [copied, setCopied] = useState(false);

  return (
    <main>
      <h1>Riskline</h1>
      <p>Paste a unified diff. The check runs in this browser.</p>
      <label htmlFor="diff">Diff</label>
      <textarea
        id="diff"
        value={diff}
        onChange={(event) => setDiff(event.target.value)}
      />
      <div>
        <button
          type="button"
          onClick={() => {
            setCopied(false);
            setResult(checkDiff(diff));
          }}
        >
          Check diff
        </button>
        {result?.kind === "card" ? (
          <button
            type="button"
            onClick={() => {
              void navigator.clipboard.writeText(result.card.pullRequest).then(() => setCopied(true));
            }}
          >
            Copy pull request
          </button>
        ) : null}
      </div>
      {copied ? <p>Copied.</p> : null}
      {result?.kind === "message" ? <p role="alert">{result.message}</p> : null}
      {result?.kind === "card" ? <pre>{result.card.markdown}</pre> : null}
    </main>
  );
}
```
